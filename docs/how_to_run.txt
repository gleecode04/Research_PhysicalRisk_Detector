# --- Train (3-class frustration, final_files_ML layout) ---
#   python train.py --feature-source final_ml [--final-ml-root <DATA>] [--output-dir <DIR>] [--final-ml-version <NAME|auto>]
#
# Placeholders:
#   <DATA>     = folder with labels.txt and Frustration/*.csv  (default: ./Data/final_files_ML)
#   <DIR>      = where metrics + joblib are written (default for final_ml: ./Data/outputs/ml_results_final_ml)
#   <VERSION>  = model run id under Data/outputs/ml_results_final_ml/versions/<VERSION>
#
# Versioning (recommended for each new training run):
#   --final-ml-version auto     → creates Data/outputs/ml_results_final_ml/versions/v1, v2, v3, …
#   --final-ml-version my_exp   → writes to Data/outputs/ml_results_final_ml/versions/my_exp
# Omit --final-ml-version to train into Data/outputs/ml_results_final_ml/ directly (may overwrite).

python train.py --feature-source final_ml --final-ml-root Data/final_files_ML --final-ml-version auto
